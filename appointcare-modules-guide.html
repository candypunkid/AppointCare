AppointCare — nWidart Module Architecture

Full-stack guide: customer fills appointment form → Laravel dispatches a job → Twilio places an AI-powered voice call → OpenAI Realtime API conducts the live conversation → appointment status auto-updates in the database.

Project Setup

Install Laravel 13, configure nWidart/laravel-modules, and set up all environment variables.

1. Create Laravel 13 project

terminalbash
composer create-project laravel/laravel appointcare "^13.0"
cd appointcare

terminalbash
composer require nwidart/laravel-modules
composer require twilio/sdk
composer require openai-php/laravel

composer require react/event-loop        # ReactPHP event loop
composer require textalk/websocket       # WebSocket client to OpenAI

Appointment Model

Modules/Appointment/App/Models/Appointment.phpphp
namespace Modules\Appointment\App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\SoftDeletes;
use Modules\Appointment\App\Events\AppointmentStatusChanged;

class Appointment extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'customer_name', 'customer_phone', 'customer_email',
        'service_type', 'notes', 'preferred_datetime',
        'confirmed_datetime', 'status', 'twilio_call_sid',
        'stream_sid', 'call_notes', 'call_attempts',
        'last_called_at', 'tenant_id',
    ];

    protected $casts = [
        'preferred_datetime'  => 'datetime',
        'confirmed_datetime'  => 'datetime',
        'last_called_at'      => 'datetime',
    ];

    // ── Helpers ────────────────────────────────────────────────────

    public function isPending(): bool
    {
        return $this->status === 'pending';
    }

    public function isCalling(): bool
    {
        return $this->status === 'calling';
    }

    public function isResolved(): bool
    {
        return in_array($this->status, ['confirmed', 'cancelled', 'rescheduled']);
    }

    /**
     * Update status and fire event — single point of status change.
     */
    public function markAs(string $status, ?string $notes = null): void
    {
        $old = $this->status;
        $this->update([
            'status'     => $status,
            'call_notes' => $notes ?? $this->call_notes,
        ]);
        event(new AppointmentStatusChanged($this, $old, $status));
    }

    public function confirm(?string $notes = null): void
    {
        $this->update(['confirmed_datetime' => $this->preferred_datetime]);
        $this->markAs('confirmed', $notes ?? 'Confirmed via AI voice call.');
    }

    public function cancel(?string $notes = null): void
    {
        $this->markAs('cancelled', $notes ?? 'Cancelled by customer via AI voice call.');
    }

    public function reschedule(string $newDatetime, ?string $notes = null): void
    {
        $this->update(['confirmed_datetime' => $newDatetime]);
        $this->markAs('rescheduled', $notes ?? "Rescheduled to {$newDatetime} via AI.");
    }
}

Appointment Controller

Modules/Appointment/App/Http/Controllers/AppointmentController.phpphp
namespace Modules\Appointment\App\Http\Controllers;

use App\Http\Controllers\Controller;
use Modules\Appointment\App\Http\Requests\StoreAppointmentRequest;
use Modules\Appointment\App\Models\Appointment;
use Modules\Appointment\App\Jobs\InitiateAICall;
use Illuminate\Http\Request;
use Illuminate\Http\JsonResponse;

class AppointmentController extends Controller
{
    /** Customer submits the appointment form */
    public function store(StoreAppointmentRequest $request): JsonResponse
    {
        $appointment = Appointment::create($request->validated());

        // Dispatch AI call job after 1 min delay (give customer time)
        InitiateAICall::dispatch($appointment)
            ->delay(now()->addMinute())
            ->onQueue('calls');

        return response()->json([
            'success' => true,
            'message' => 'Appointment submitted! You will receive a call shortly.',
            'data'    => [
                'id'     => $appointment->id,
                'status' => $appointment->status,
            ],
        ], 201);
    }

    /** Admin list all appointments */
    public function index(Request $request): JsonResponse
    {
        $appointments = Appointment::query()
            ->when($request->status, fn($q, $s) => $q->where('status', $s))
            ->when($request->date, fn($q, $d) => $q->whereDate('preferred_datetime', $d))
            ->latest()
            ->paginate(20);

        return response()->json($appointments);
    }

    /** Admin manually update status */
    public function updateStatus(Appointment $appointment, Request $request): JsonResponse
    {
        $request->validate([
            'status' => 'required|in:confirmed,cancelled,rescheduled,pending',
            'notes'  => 'nullable|string',
        ]);

        $appointment->markAs($request->status, $request->notes);

        return response()->json([
            'success' => true,
            'data'    => $appointment->fresh(),
        ]);
    }

    /** Admin manually trigger a call retry */
    public function retryCall(Appointment $appointment): JsonResponse
    {
        abort_if($appointment->call_attempts >= 5, 422, 'Max call attempts reached.');

        $appointment->update(['status' => 'pending']);
        InitiateAICall::dispatch($appointment)->onQueue('calls');

        return response()->json(['success' => true, 'message' => 'Call queued.']);
    }
}

Observer (auto-log status changes)

namespace Modules\Appointment\App\Observers;

use Modules\Appointment\App\Models\Appointment;
use Illuminate\Support\Facades\Log;

class AppointmentObserver
{
    public function updated(Appointment $appointment): void
    {
        if ($appointment->wasChanged('status')) {
            Log::channel('appointments')->info('Status changed', [
                'id'     => $appointment->id,
                'from'   => $appointment->getOriginal('status'),
                'to'     => $appointment->status,
                'phone'  => $appointment->customer_phone,
            ]);

            // Broadcast to admin dashboard in real-time
            broadcast(new \Modules\Appointment\App\Events\AppointmentStatusChanged(
                $appointment,
                $appointment->getOriginal('status'),
                $appointment->status
            ))->toOthers();
        }
    }
}

Twilio Service

Modules/Twilio/App/Services/TwilioService.phpphp
namespace Modules\Twilio\App\Services;

use Twilio\Rest\Client;
use Twilio\TwiML\VoiceResponse;
use Modules\Appointment\App\Models\Appointment;
use Illuminate\Support\Facades\Log;

class TwilioService
{
    protected Client $client;

    public function __construct()
    {
        $this->client = new Client(
            config('services.twilio.sid'),
            config('services.twilio.token')
        );
    }

    /**
     * Place outbound call to customer.
     * Twilio will hit our /twilio/voice webhook when answered.
     */
    public function callCustomer(Appointment $appointment): string
    {
        $webhookBase = config('services.twilio.webhook_base');

        $call = $this->client->calls->create(
            $appointment->customer_phone,
            config('services.twilio.from'),
            [
                'url'    => $webhookBase . '/twilio/voice?appointment_id=' . $appointment->id,
                'method' => 'POST',
                'statusCallback'      => $webhookBase . '/twilio/status?appointment_id=' . $appointment->id,
                'statusCallbackEvent' => ['initiated', 'ringing', 'answered', 'completed'],
                'statusCallbackMethod' => 'POST',
                'timeout' => 30,
                'machineDetection' => 'Enable',  // Auto-detect voicemail
            ]
        );

        $appointment->update([
            'twilio_call_sid' => $call->sid,
            'status'          => 'calling',
            'call_attempts'   => $appointment->call_attempts + 1,
            'last_called_at'  => now(),
        ]);

        Log::info('Call initiated', ['sid' => $call->sid, 'to' => $appointment->customer_phone]);

        return $call->sid;
    }

    /** Build TwiML: greeting + media stream to our WebSocket */
    public function buildVoiceTwiML(Appointment $appointment): string
    {
        $response = new VoiceResponse();

        // Greeting before connecting the AI
        $response->say(
            "Hello, {$appointment->customer_name}. "
            . "This is AppointCare calling about your {$appointment->service_type} appointment. "
            . "Please stay on the line, our AI assistant will be right with you.",
            ['voice' => 'Polly.Amy', 'language' => 'en-GB']
        );

        // Connect Twilio media stream to our WebSocket server
        $connect = $response->connect();
        $stream  = $connect->stream([
            'url'  => 'wss://' . parse_url(config('services.twilio.webhook_base'), PHP_URL_HOST)
                     . '/ws/aicall/' . $appointment->id,
            'name' => 'openai-stream',
        ]);
        $stream->parameter(['name' => 'appointmentId', 'value' => (string) $appointment->id]);

        return (string) $response;
    }

    /** Send SMS to customer */
    public function sendSms(string $to, string $message): void
    {
        $this->client->messages->create($to, [
            'from' => config('services.twilio.from'),
            'body' => $message,
        ]);
    }
}

Twilio Controller (Webhooks)

Modules/Twilio/App/Http/Controllers/TwilioWebhookController.phpphp
namespace Modules\Twilio\App\Http\Controllers;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Http\Response;
use Modules\Appointment\App\Models\Appointment;
use Modules\Appointment\App\Jobs\InitiateAICall;
use Modules\Twilio\App\Services\TwilioService;

class TwilioWebhookController extends Controller
{
    public function __construct(protected TwilioService $twilio) {}

    /**
     * Twilio calls this when the customer answers.
     * We return TwiML that connects them to the AI.
     */
    public function voice(Request $request): Response
    {
        $appointment = Appointment::findOrFail($request->query('appointment_id'));

        // Handle answering machine detection
        if ($request->input('AnsweredBy') === 'machine_start') {
            return $this->voicemailResponse($appointment);
        }

        $twiml = $this->twilio->buildVoiceTwiML($appointment);

        return response($twiml, 200)
            ->header('Content-Type', 'text/xml');
    }

    private function voicemailResponse(Appointment $appointment): Response
    {
        use Twilio\TwiML\VoiceResponse;
        $r = new VoiceResponse();
        $r->say("Hello {$appointment->customer_name}, this is AppointCare. "
              . "Please call us back or check your email to confirm your appointment. Goodbye.");
        $r->hangup();
        $appointment->markAs('no_answer', 'Voicemail detected.');
        return response((string)$r, 200)->header('Content-Type', 'text/xml');
    }

    /**
     * Twilio status callback: completed, no-answer, busy, failed
     */
    public function statusCallback(Request $request): Response
    {
        $callSid = $request->input('CallSid');
        $callStatus = $request->input('CallStatus');
        $appointment = Appointment::where('twilio_call_sid', $callSid)->first();

        if (!$appointment) {
            return response('OK', 200);
        }

        match ($callStatus) {
            'no-answer', 'busy' => $this->scheduleRetry($appointment),
            'failed'             => $appointment->markAs('failed', 'Twilio call failed.'),
            'completed'          => null,  // AI handled status via function calling
            default              => null,
        };

        return response('OK', 200);
    }

    private function scheduleRetry(Appointment $appointment): void
    {
        if ($appointment->call_attempts >= 3) {
            $appointment->markAs('no_answer', 'Maximum call attempts reached.');
            return;
        }
        // Retry in 1 hour
        $appointment->update(['status' => 'pending']);
        InitiateAICall::dispatch($appointment)
            ->delay(now()->addHour())
            ->onQueue('calls');
    }
}

Twilio Webhook Signature Middleware

Modules/Twilio/App/Http/Middleware/ValidateTwilioWebhook.phpphp
namespace Modules\Twilio\App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Twilio\Security\RequestValidator;

class ValidateTwilioWebhook
{
    public function handle(Request $request, Closure $next): mixed
    {
        // Skip in local/testing environment
        if (app()->environment(['local', 'testing'])) {
            return $next($request);
        }

        $validator = new RequestValidator(config('services.twilio.token'));

        $valid = $validator->validate(
            $request->header('X-Twilio-Signature', ''),
            $request->fullUrl(),
            $request->post()
        );

        abort_unless($valid, 403, 'Invalid Twilio webhook signature.');

        return $next($request);
    }
}

AICall Module — WebSocket + OpenAI

The AICall module bridges Twilio's raw audio stream with OpenAI's Realtime API over WebSocket. Audio flows in both directions, creating a live voice conversation.

ℹ
Twilio streams audio as G.711 μ-law (PCMU) at 8kHz. OpenAI Realtime API accepts this format directly — no transcoding needed. Set input_audio_format and output_audio_format both to g711_ulaw.

OpenAI Realtime Service

Modules/AICall/App/Services/OpenAIRealtimeService.phpphp
namespace Modules\AICall\App\Services;

use WebSocket\Client as WsClient;
use Modules\Appointment\App\Models\Appointment;

class OpenAIRealtimeService
{
    public function createClient(Appointment $appointment): WsClient
    {
        $model  = config('aicall.model');
        $apiKey = config('openai.api_key');

        $client = new WsClient(
            "wss://api.openai.com/v1/realtime?model={$model}",
            [
                'headers' => [
                    'Authorization' => "Bearer {$apiKey}",
                    'OpenAI-Beta'   => 'realtime=v1',
                ],
                'timeout' => 30,
            ]
        );

        // Send session configuration immediately after connect
        $client->send(json_encode($this->buildSessionConfig($appointment)));

        return $client;
    }

    private function buildSessionConfig(Appointment $appointment): array
    {
        $name     = $appointment->customer_name;
        $service  = $appointment->service_type;
        $datetime = $appointment->preferred_datetime->format('l, F j \a\t g:i A');

        $instructions = config('aicall.system_prompt')
            . "\n\nCurrent appointment details:"
            . "\n- Customer: {$name}"
            . "\n- Service: {$service}"
            . "\n- Scheduled time: {$datetime}"
            . "\n- Appointment ID: {$appointment->id}";

        return [
            'type'    => 'session.update',
            'session' => [
                'modalities'          => ['audio', 'text'],
                'instructions'        => $instructions,
                'voice'               => config('aicall.voice'),
                'input_audio_format'  => 'g711_ulaw',
                'output_audio_format' => 'g711_ulaw',
                'input_audio_transcription' => ['model' => 'whisper-1'],
                'turn_detection'      => [
                    'type'              => 'server_vad',
                    'threshold'         => 0.5,
                    'prefix_padding_ms' => 300,
                    'silence_duration_ms' => 800,
                ],
                'tools'               => $this->getTools(),
                'tool_choice'         => 'auto',
                'temperature'         => 0.7,
            ],
        ];
    }

    public function getTools(): array
    {
        return [
            [
                'type'        => 'function',
                'name'        => 'confirm_appointment',
                'description' => 'Call when the customer confirms their appointment.',
                'parameters'  => ['type' => 'object', 'properties' => []],
            ],
            [
                'type'        => 'function',
                'name'        => 'cancel_appointment',
                'description' => 'Call when the customer wants to cancel.',
                'parameters'  => [
                    'type'       => 'object',
                    'properties' => [
                        'reason' => ['type' => 'string', 'description' => 'Cancellation reason if provided'],
                    ],
                ],
            ],
            [
                'type'        => 'function',
                'name'        => 'reschedule_appointment',
                'description' => 'Call when the customer wants to reschedule to a new time.',
                'parameters'  => [
                    'type'       => 'object',
                    'properties' => [
                        'new_datetime' => [
                            'type'        => 'string',
                            'description' => 'New preferred datetime in ISO 8601 format',
                        ],
                        'reason' => ['type' => 'string'],
                    ],
                    'required' => ['new_datetime'],
                ],
            ],
            [
                'type'        => 'function',
                'name'        => 'end_call',
                'description' => 'Call when conversation is complete and it is time to say goodbye.',
                'parameters'  => ['type' => 'object', 'properties' => []],
            ],
        ];
    }
}

Status Change Listener

Modules/Notification/App/Listeners/SendAppointmentNotification.phpphp
namespace Modules\Notification\App\Listeners;

use Modules\Appointment\App\Events\AppointmentStatusChanged;
use Modules\Notification\App\Notifications\AppointmentStatusNotification;
use Modules\Twilio\App\Services\TwilioService;
use Illuminate\Support\Facades\Notification;

class SendAppointmentNotification
{
    public function __construct(protected TwilioService $twilio) {}

    public function handle(AppointmentStatusChanged $event): void
    {
        $apt = $event->appointment;
        $resolved = ['confirmed', 'cancelled', 'rescheduled'];

        if (!in_array($apt->status, $resolved)) {
            return;
        }

        // Send email if email provided
        if ($apt->customer_email) {
            Notification::route('mail', $apt->customer_email)
                ->notify(new AppointmentStatusNotification($apt));
        }

        // Send SMS via Twilio
        $smsText = match ($apt->status) {
            'confirmed'   => "✅ AppointCare: Your {$apt->service_type} appointment on "
                            . $apt->confirmed_datetime?->format('M j \a\t g:i A') . " is confirmed!",
            'cancelled'   => "❌ AppointCare: Your {$apt->service_type} appointment has been cancelled.",
            'rescheduled' => "🔄 AppointCare: Rescheduled to "
                            . $apt->confirmed_datetime?->format('M j \a\t g:i A') . ".",
            default       => null,
        };

        if ($smsText) {
            $this->twilio->sendSms($apt->customer_phone, $smsText);
        }
    }
}

namespace Modules\Appointment\App\Events;

use Illuminate\Broadcasting\Channel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;
use Modules\Appointment\App\Models\Appointment;

class AppointmentStatusChanged implements ShouldBroadcast
{
    use Dispatchable, SerializesModels;

    public function __construct(
        public readonly Appointment $appointment,
        public readonly string $oldStatus,
        public readonly string $newStatus,
    ) {}

    public function broadcastOn(): array
    {
        return [new Channel('appointments')];
    }

    public function broadcastAs(): string
    {
        return 'appointment.status.changed';
    }

    public function broadcastWith(): array
    {
        return [
            'id'         => $this->appointment->id,
            'name'       => $this->appointment->customer_name,
            'service'    => $this->appointment->service_type,
            'old_status' => $this->oldStatus,
            'new_status' => $this->newStatus,
            'notes'      => $this->appointment->call_notes,
        ];
    }
}

InitiateAICall Job

Modules/Appointment/App/Jobs/InitiateAICall.phpphp
namespace Modules\Appointment\App\Jobs;

use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Modules\Appointment\App\Models\Appointment;
use Modules\Twilio\App\Services\TwilioService;
use Illuminate\Support\Facades\Log;

class InitiateAICall implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries   = 3;
    public int $timeout = 60;
    public int $backoff  = 30;   // seconds between retries

    public function __construct(public readonly Appointment $appointment) {}

    public function handle(TwilioService $twilio): void
    {
        $fresh = $this->appointment->fresh();

        // Guard: only call if still pending
        if (!$fresh->isPending()) {
            Log::info('Skipping call — appointment no longer pending', ['id' => $fresh->id]);
            return;
        }

        try {
            $twilio->callCustomer($fresh);
        } catch (\Exception $e) {
            Log::error('InitiateAICall failed', [
                'appointment_id' => $fresh->id,
                'error'          => $e->getMessage(),
            ]);
            $this->fail($e);
        }
    }

    public function failed(\Throwable $exception): void
    {
        $this->appointment->markAs(
            'failed',
            'Call job permanently failed: ' . $exception->getMessage()
        );
    }

    /** Prevent duplicate calls for same appointment */
    public function uniqueId(): string
    {
        return 'ai-call-' . $this->appointment->id;
    }
}

use Illuminate\Support\Facades\Schedule;
use Modules\Appointment\App\Models\Appointment;
use Modules\Appointment\App\Jobs\InitiateAICall;

// ── Every 5 minutes: Call any new pending appointments ─────────
Schedule::call(function () {
    Appointment::query()
        ->where('status', 'pending')
        ->where('call_attempts', 0)
        ->where('created_at', '<=', now()->subMinutes(2)) // 2min grace
        ->each(function (Appointment $apt) {
            InitiateAICall::dispatch($apt)->onQueue('calls');
        });
})
->everyFiveMinutes()
->name('call-new-pending-appointments')
->withoutOverlapping();

// ── Hourly: Retry failed/no-answer appointments ───────────────
Schedule::call(function () {
    Appointment::query()
        ->where('status', 'pending')
        ->where('call_attempts', '>', 0)
        ->where('call_attempts', '<', 3)
        ->where('last_called_at', '<=', now()->subHour())
        ->each(function (Appointment $apt) {
            InitiateAICall::dispatch($apt)->onQueue('calls');
        });
})
->hourly()
->name('retry-unanswered-calls')
->withoutOverlapping();

// ── Every 15min: Mark calling→pending if stuck > 10min ───────
Schedule::call(function () {
    Appointment::where('status', 'calling')
        ->where('last_called_at', '<=', now()->subMinutes(10))
        ->update(['status' => 'pending']);
})
->everyFifteenMinutes()
->name('reset-stuck-calling-status');

// ── Daily 8am: Send daily summary to admin ────────────────────
Schedule::job(new \Modules\Dashboard\App\Jobs\SendDailyReport)

    ->name('daily-admin-report')

10 · WebSocket Server

WebSocket Server (Ratchet)

The WebSocket server runs as a long-lived PHP process. It accepts incoming connections from Twilio's media stream and bridges them to OpenAI's Realtime API WebSocket.

Artisan command to start server

Modules/AICall/App/Console/Commands/StartWebSocketServer.phpphp
namespace Modules\AICall\App\Console\Commands;

use Illuminate\Console\Command;
use Ratchet\Server\IoServer;
use Ratchet\Http\HttpServer;
use Ratchet\WebSocket\WsServer;
use Modules\AICall\App\WebSocket\AICallHandler;

class StartWebSocketServer extends Command
{
    protected $signature   = 'aicall:websocket-server {--port=8090}';
    protected $description = 'Start the AICall WebSocket server for Twilio media streams';

    public function handle(): void
    {
        $port = (int) $this->option('port');
        $this->info("Starting AICall WebSocket server on port {$port}...");

        $server = IoServer::factory(
            new HttpServer(
                new WsServer(
                    new AICallHandler()
                )
            ),
            $port,
            '0.0.0.0'
        );

        $this->info("WebSocket server running. Waiting for Twilio connections...");
        $server->run();
    }
}

11 · OpenAI Realtime

AICallHandler — Full WebSocket Bridge

This is the core bridge class. Each WebSocket connection from Twilio gets its own OpenAI Realtime connection. Audio flows in both directions simultaneously.

Modules/AICall/App/WebSocket/AICallHandler.phpphp
namespace Modules\AICall\App\WebSocket;

use Ratchet\MessageComponentInterface;
use Ratchet\ConnectionInterface;
use WebSocket\Client as WsClient;
use Modules\Appointment\App\Models\Appointment;
use Modules\AICall\App\Services\OpenAIRealtimeService;
use Modules\AICall\App\WebSocket\FunctionCallHandler;
use Illuminate\Support\Facades\Log;

class AICallHandler implements MessageComponentInterface
{
    /** Active sessions keyed by Ratchet connection resource ID */
    protected array $sessions = [];

    protected OpenAIRealtimeService $openaiService;
    protected FunctionCallHandler $functionHandler;

    public function __construct()
    {
        $this->openaiService   = app(OpenAIRealtimeService::class);
        $this->functionHandler = app(FunctionCallHandler::class);
    }

    /** Twilio opens a WebSocket — extract appointment ID and connect to OpenAI */
    public function onOpen(ConnectionInterface $conn): void
    {
        // URI format: /ws/aicall/{appointmentId}
        $path          = $conn->httpRequest->getUri()->getPath();
        $appointmentId = (int) basename($path);
        $appointment   = Appointment::find($appointmentId);

        if (!$appointment) {
            Log::warning('AICall: appointment not found', ['id' => $appointmentId]);
            $conn->close();
            return;
        }

        try {
            $openaiWs = $this->openaiService->createClient($appointment);
        } catch (\Exception $e) {
            Log::error('OpenAI WebSocket connection failed', ['error' => $e->getMessage()]);
            $conn->close();
            return;
        }

        $this->sessions[$conn->resourceId] = [
            'twilio'      => $conn,
            'openai'      => $openaiWs,
            'appointment' => $appointment,
            'stream_sid'  => null,
            'transcript'  => [],
        ];

        Log::info('AICall session opened', [
            'appointment_id' => $appointmentId,
            'customer'       => $appointment->customer_name,
        ]);
    }

    /** Handle incoming Twilio WebSocket messages */
    public function onMessage(ConnectionInterface $from, $msg): void
    {
        $session = $this->sessions[$from->resourceId] ?? null;
        if (!$session) return;

        $data = json_decode($msg, true);
        $event = $data['event'] ?? null;

        match ($event) {
            'start'  => $this->handleStart($from->resourceId, $data),
            'media'  => $this->handleMedia($from->resourceId, $data),
            'stop'   => $this->handleStop($from->resourceId),
            default  => null,
        };

        // Poll OpenAI for any pending responses
        $this->pollOpenAI($from->resourceId);
    }

    private function handleStart(int $resourceId, array $data): void
    {
        $this->sessions[$resourceId]['stream_sid'] = $data['streamSid'] ?? null;
        $appointment = $this->sessions[$resourceId]['appointment'];
        $appointment->update(['stream_sid' => $data['streamSid'] ?? null]);

        Log::info('AICall media stream started', ['stream_sid' => $data['streamSid']]);
    }

    private function handleMedia(int $resourceId, array $data): void
    {
        $session  = $this->sessions[$resourceId];
        $openaiWs = $session['openai'];

        // Forward Twilio audio to OpenAI
        $openaiWs->send(json_encode([
            'type'  => 'input_audio_buffer.append',
            'audio' => $data['media']['payload'], // base64 G.711 ulaw audio
        ]));
    }

    private function handleStop(int $resourceId): void
    {
        Log::info('Twilio stream stopped', ['resource' => $resourceId]);
        $this->closeSession($resourceId);
    }

    /** Poll OpenAI for responses and handle them */
    private function pollOpenAI(int $resourceId): void
    {
        $session  = $this->sessions[$resourceId] ?? null;
        if (!$session) return;

        $openaiWs   = $session['openai'];
        $twilioConn = $session['twilio'];
        $streamSid  = $session['stream_sid'];

        try {
            $raw = $openaiWs->receive();
            if (!$raw) return;

            $event = json_decode($raw, true);
            $type  = $event['type'] ?? '';

            match (true) {
                // Stream AI audio back to Twilio customer
                $type === 'response.audio.delta' => $twilioConn->send(json_encode([
                    'event'     => 'media',
                    'streamSid' => $streamSid,
                    'media'     => ['payload' => $event['delta']],
                ])),

                // Mark audio buffer as committed when AI stops speaking
                $type === 'response.audio.done' => $twilioConn->send(json_encode([
                    'event'     => 'mark',
                    'streamSid' => $streamSid,
                    'mark'      => ['name' => 'ai-done'],
                ])),

                // Handle function calls (confirm/cancel/reschedule)
                $type === 'response.function_call_arguments.done'
                    => $this->handleFunctionCall($resourceId, $event),

                // Store transcript segments
                $type === 'conversation.item.input_audio_transcription.completed'
                    => $this->storeTranscript($resourceId, 'customer', $event['transcript'] ?? ''),

                $type === 'response.text.done'
                    => $this->storeTranscript($resourceId, 'ai', $event['text'] ?? ''),

                // Log errors
                $type === 'error' => Log::error('OpenAI Realtime error', $event),

                default => null,
            };
        } catch (\Exception $e) {
            // Non-blocking — timeout is expected when no data
        }
    }

    private function handleFunctionCall(int $resourceId, array $event): void
    {
        $session     = $this->sessions[$resourceId];
        $appointment = $session['appointment'];
        $name        = $event['name'];
        $args        = json_decode($event['arguments'] ?? '{}', true);

        $this->functionHandler->handle($name, $args, $appointment);

        // Tell OpenAI the function completed so it can continue the conversation
        $session['openai']->send(json_encode([
            'type'    => 'conversation.item.create',
            'item'    => [
                'type'        => 'function_call_output',
                'call_id'    => $event['call_id'],
                'output'     => json_encode(['success' => true]),
            ],
        ]));

        // Tell OpenAI to generate a verbal confirmation
        $session['openai']->send(json_encode(['type' => 'response.create']));
    }

    private function storeTranscript(int $resourceId, string $speaker, string $text): void
    {
        $this->sessions[$resourceId]['transcript'][] = [
            'speaker' => $speaker,
            'text'    => $text,
            'time'    => now()->toTimeString(),
        ];
    }

    public function onClose(ConnectionInterface $conn): void
    {
        $session = $this->sessions[$conn->resourceId] ?? null;
        if ($session) {
            // Save full transcript to call_notes
            $lines = array_map(
                fn($t) => "[{$t['time']}] {$t['speaker']}: {$t['text']}",
                $session['transcript']
            );
            $session['appointment']->update([
                'call_notes' => implode("\n", $lines),
            ]);
        }
        $this->closeSession($conn->resourceId);
    }

    public function onError(ConnectionInterface $conn, \Exception $e): void
    {
        Log::error('AICallHandler WebSocket error: ' . $e->getMessage());
        $conn->close();
    }

    private function closeSession(int $resourceId): void
    {
        if (isset($this->sessions[$resourceId])) {
            try {
                $this->sessions[$resourceId]['openai']->close();
            } catch (\Exception $e) {}
            unset($this->sessions[$resourceId]);
        }
    }
}
12 · Function Calling

Function Call Handler

When OpenAI decides the customer has confirmed, cancelled, or wants to reschedule, it calls one of the registered functions. This handler maps those calls to Laravel model actions.

Modules/AICall/App/WebSocket/FunctionCallHandler.phpphp
namespace Modules\AICall\App\WebSocket;

use Modules\Appointment\App\Models\Appointment;
use Illuminate\Support\Facades\Log;

class FunctionCallHandler
{
    /**
     * Dispatch the correct appointment action based on OpenAI function name.
     *
     * @param  string      $functionName  confirm_appointment | cancel_appointment | reschedule_appointment | end_call
     * @param  array       $args          Arguments from OpenAI function call
     * @param  Appointment $appointment   The appointment being managed
     */
    public function handle(string $functionName, array $args, Appointment $appointment): void
    {
        Log::info('Function called by OpenAI', [
            'function'       => $functionName,
            'args'           => $args,
            'appointment_id' => $appointment->id,
        ]);

        match ($functionName) {
            'confirm_appointment'    => $this->confirm($appointment),
            'cancel_appointment'     => $this->cancel($appointment, $args),
            'reschedule_appointment' => $this->reschedule($appointment, $args),
            'end_call'               => $this->endCall($appointment),
            default                  => Log::warning('Unknown function: ' . $functionName),
        };
    }

    private function confirm(Appointment $appointment): void
    {
        if ($appointment->isResolved()) return;

        $appointment->confirm('Customer confirmed via AI voice call.');

        Log::info('Appointment confirmed by AI', ['id' => $appointment->id]);
    }

    private function cancel(Appointment $appointment, array $args): void
    {
        if ($appointment->isResolved()) return;

        $reason = $args['reason'] ?? 'No reason provided.';
        $appointment->cancel("Cancelled via AI voice call. Reason: {$reason}");

        Log::info('Appointment cancelled by AI', ['id' => $appointment->id, 'reason' => $reason]);
    }

    private function reschedule(Appointment $appointment, array $args): void
    {
        if ($appointment->isResolved()) return;

        $newDatetime = $args['new_datetime'] ?? null;

        if (!$newDatetime) {
            Log::warning('Reschedule called without new_datetime', $args);
            return;
        }

        try {
            $parsed = \Carbon\Carbon::parse($newDatetime);
        } catch (\Exception $e) {
            Log::error('Invalid datetime from OpenAI', ['value' => $newDatetime]);
            return;
        }

        $appointment->reschedule(
            $parsed->toDateTimeString(),
            "Rescheduled to {$parsed->format('D M j \a\t g:i A')} via AI voice call."
        );

        Log::info('Appointment rescheduled by AI', ['id' => $appointment->id, 'new' => $parsed]);
    }

    private function endCall(Appointment $appointment): void
    {
        // If still unresolved after conversation, log it
        if (!$appointment->fresh()->isResolved()) {
            Log::info('Call ended without resolution', ['id' => $appointment->id]);
        }
    }
}

Production Deployment

Full deployment checklist

1
Optimize Laravel for production
php artisan config:cache && php artisan route:cache && php artisan view:cache && php artisan event:cache
2
Run module migrations
php artisan module:migrate --force — runs migrations from all enabled modules.
3
Start Supervisor
supervisorctl reread && supervisorctl update && supervisorctl start all — starts queue workers, WebSocket server, and Reverb.
4
Add cron entry
* * * * * cd /var/www/appointcare && php artisan schedule:run >> /dev/null 2>&1
5
Configure Twilio webhook URLs
In Twilio dashboard → Voice → your number → set Voice URL to https://your-domain.com/twilio/voice and Status Callback URL to /twilio/status.
6
Nginx — proxy all WebSocket routes
WebSocket server (port 8090 for Twilio audio) and Reverb (port 8080 for broadcasting) both need WSS proxy in Nginx.
7
Local dev: use ngrok
ngrok http 8000 — update TWILIO_WEBHOOK_BASE_URL and APP_URL in .env to the ngrok HTTPS URL.
